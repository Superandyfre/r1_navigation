cmake_minimum_required(VERSION 3.8)
project(r1_neupan_runtime)
find_package(ament_cmake REQUIRED)
find_package(nlohmann_json REQUIRED)
install(DIRECTORY include/ DESTINATION include)
ament_export_include_directories(include)
ament_export_dependencies(nlohmann_json)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_motion_authorization test/test_motion_authorization.cpp)
  target_compile_features(test_motion_authorization PUBLIC cxx_std_17)
  target_compile_definitions(test_motion_authorization PRIVATE
    R1_SOURCE_VECTORS_PATH="${CMAKE_CURRENT_SOURCE_DIR}/test/source_protocol_vectors.json")
  target_include_directories(test_motion_authorization PRIVATE include)
  target_link_libraries(test_motion_authorization nlohmann_json::nlohmann_json)
  ament_add_gtest(test_controller_execution test/test_controller_execution.cpp)
  target_compile_features(test_controller_execution PUBLIC cxx_std_17)
  target_include_directories(test_controller_execution PRIVATE include)
  target_link_libraries(test_controller_execution nlohmann_json::nlohmann_json)
  ament_add_gtest(test_stopped_odometry test/test_stopped_odometry.cpp)
  target_compile_features(test_stopped_odometry PUBLIC cxx_std_17)
  target_include_directories(test_stopped_odometry PRIVATE include)
  ament_add_gtest(test_command_barrier test/test_command_barrier.cpp)
  target_compile_features(test_command_barrier PUBLIC cxx_std_17)
  target_include_directories(test_command_barrier PRIVATE include)
endif()
ament_package()
